cmake_minimum_required(VERSION 3.24)
project(CleanFlow VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
qt_standard_project_setup()
qt_add_executable(CleanFlow app/main.cpp src/core/JobManager.cpp)
qt_add_qml_module(CleanFlow URI CleanFlow VERSION 1.0 QML_FILES app/qml/Main.qml)
target_include_directories(CleanFlow PRIVATE src)
target_link_libraries(CleanFlow PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
